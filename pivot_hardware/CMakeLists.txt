cmake_minimum_required(VERSION 3.16)
project(pivot_hardware)
find_package(ament_cmake REQUIRED)
find_package(gz_ros2_control REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(hardware_interface REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
add_library(pivot_motor_system SHARED src/motor_system.cpp)
target_compile_features(pivot_motor_system PUBLIC cxx_std_17)
target_compile_options(pivot_motor_system PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(pivot_motor_system PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
ament_target_dependencies(pivot_motor_system gz_ros2_control hardware_interface pluginlib rclcpp)
target_link_libraries(pivot_motor_system gz-sim8::gz-sim8)
pluginlib_export_plugin_description_file(gz_ros2_control motor_plugins.xml)
install(TARGETS pivot_motor_system LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_motor_model test/test_motor_model.cpp)
  target_include_directories(test_motor_model PRIVATE include)
  target_compile_features(test_motor_model PRIVATE cxx_std_17)
  ament_add_gtest(test_motor_system test/test_motor_system.cpp)
  target_include_directories(test_motor_system PRIVATE include)
  target_compile_features(test_motor_system PRIVATE cxx_std_17)
  ament_target_dependencies(test_motor_system gz_ros2_control hardware_interface pluginlib rclcpp)
  target_link_libraries(test_motor_system gz-sim8::gz-sim8)
endif()
ament_package()
